cmake_minimum_required(VERSION 3.25)
project(PrismestraPawnIOLibSourceBuild LANGUAGES CXX RC VERSION 2.2.0)
set(PAWN_SOURCE "" CACHE PATH "Extracted namazso/PawnIO source root (pinned by dependency-lock.json)")
if(NOT EXISTS "${PAWN_SOURCE}/PawnIOLib/PawnIOLib.cpp")
    message(FATAL_ERROR "Pass -DPAWN_SOURCE:PATH=<extracted pinned PawnIO source root>")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_library(PawnIOLib SHARED "${PAWN_SOURCE}/PawnIOLib/PawnIOLib.cpp" "${PAWN_SOURCE}/PawnIOLib/resource.rc")
target_include_directories(PawnIOLib PRIVATE "${PAWN_SOURCE}/PawnIOLib/include" "${PAWN_SOURCE}/PawnIO/include")
target_compile_definitions(PawnIOLib PRIVATE PAWNIO_NAME="PawnIO" PAWNIO_NAME_FULL="PawnIO local source build" PAWNIO_AUTHOR="namazso" PAWNIO_VERSION_STRING="2.2.0" PAWNIO_VERSION_STRING4="2.2.0.0" PAWNIO_MAJOR=2 PAWNIO_MINOR=2 PAWNIO_PATCH=0)
target_compile_options(PawnIOLib PRIVATE /W4 /WX /utf-8 /d1nodatetime /Brepro /Zi /EHsc)
target_link_libraries(PawnIOLib PRIVATE ntdll.lib)
target_link_options(PawnIOLib PRIVATE /ENTRY:DllEntry /DEBUG /OPT:REF /OPT:ICF=10 /FILEALIGN:0x1000 /CETCOMPAT /BREPRO /NOVCFEATURE /NOCOFFGRPINFO /PDBALTPATH:%_PDB%)
